import { useState, useRef, useCallback, useEffect } from "react";
import { Button } from "@/components/ui/button";
import { Card } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import { Progress } from "@/components/ui/progress";
import { Mic, Pause, Play, Square, Download, Trash2 } from "lucide-react";
import { toast } from "sonner";

/* ---------- Types ---------- */
interface ApiSentiment {
  label: "positive" | "negative" | "neutral";
  scores: { positive: number; negative: number; neutral: number };
}

interface ApiProsody {
  rms_energy: number; // 0..~1 (RMS)
  pitch_mean_hz: number;
  pitch_std_hz: number;
  speech_rate_frames_per_sec: number;
  tone_guess: string; // "excited / possibly angry", "neutral", "sad / calm"
}

interface ApiResponse {
  language: string; // "en", "hi", ...
  transcript: string;
  sentiment: ApiSentiment;
  prosody: ApiProsody;
  segments: Array<{ start: number; end: number; text: string }>;
}

interface VoiceAnalysis {
  emotion: string; // mapped from tone_guess
  confidence: number; // top sentiment prob
  sentiment: "positive" | "negative" | "neutral";
  energy: number; // 0..100 (derived from RMS)
  timestamp: Date;

  // extras for UI
  transcript?: string;
  language?: string;
  prosody?: ApiProsody;
  sentimentScores?: ApiSentiment["scores"];
}

interface Recording {
  id: string;
  blob: Blob;
  url: string;
  duration: number;
  analysis: VoiceAnalysis;
  timestamp: Date;
}

export const VoiceAnalyzer = () => {
  const [isRecording, setIsRecording] = useState(false);
  const [isPlaying, setIsPlaying] = useState(false);
  const [isAnalyzing, setIsAnalyzing] = useState(false);
  const [recordings, setRecordings] = useState<Recording[]>([]);
  const [currentAnalysis, setCurrentAnalysis] = useState<VoiceAnalysis | null>(
    null
  );
  const [recordingTime, setRecordingTime] = useState(0);
  const [audioLevel, setAudioLevel] = useState(0);

  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const audioContextRef = useRef<AudioContext | null>(null);
  const analyserRef = useRef<AnalyserNode | null>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const recordingTimerRef = useRef<number | null>(null);
  const audioPlayerRef = useRef<HTMLAudioElement | null>(null);

  /* ---------- Helpers ---------- */

  // scale RMS (â‰ˆ0..~0.5+) to a 0..100 UI bar
  const energyPercent = (rms: number) => {
    const pct = Math.max(0, Math.min(1, rms / 0.2)) * 100; // clamp; 0.2 ~ â€œloud speechâ€
    return pct;
  };

  // Encode Float32 PCM -> WAV 16k mono
  const floatTo16BitWav = (samples: Float32Array, sampleRate: number): Blob => {
    const bytesPerSample = 2;
    const blockAlign = bytesPerSample;
    const byteRate = sampleRate * blockAlign;
    const dataSize = samples.length * bytesPerSample;
    const buffer = new ArrayBuffer(44 + dataSize);
    const view = new DataView(buffer);

    let o = 0;
    const wS = (s: string) => { for (let i = 0; i < s.length; i++) view.setUint8(o++, s.charCodeAt(i)); };
    const w32 = (v: number) => { view.setUint32(o, v, true); o += 4; };
    const w16 = (v: number) => { view.setUint16(o, v, true); o += 2; };

    wS("RIFF"); w32(36 + dataSize); wS("WAVE");
    wS("fmt "); w32(16); w16(1); w16(1); w32(sampleRate); w32(byteRate); w16(blockAlign); w16(16);
    wS("data"); w32(dataSize);
    for (let i = 0; i < samples.length; i++) {
      const s = Math.max(-1, Math.min(1, samples[i]));
      view.setInt16(44 + i * 2, s < 0 ? s * 0x8000 : s * 0x7fff, true);
    }
    return new Blob([buffer], { type: "audio/wav" });
  };

  // Transcode WebM/Opus -> 16k mono WAV in browser
  const webmToWav16k = useCallback(async (webm: Blob): Promise<{ wav: Blob; rms: number }> => {
    const arrayBuffer = await webm.arrayBuffer();
    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    const decoded = await ctx.decodeAudioData(arrayBuffer.slice(0));

    const len = decoded.length;
    const sr = decoded.sampleRate;
    const ch = decoded.numberOfChannels;
    const mono = new Float32Array(len);
    const chans: Float32Array[] = [];
    for (let c = 0; c < ch; c++) chans.push(decoded.getChannelData(c));
    for (let i = 0; i < len; i++) {
      let s = 0; for (let c = 0; c < ch; c++) s += chans[c][i] || 0; mono[i] = s / ch;
    }

    const target = 16000;
    const frames = Math.ceil(decoded.duration * target);
    const oac = new OfflineAudioContext(1, frames, target);
    const buf = oac.createBuffer(1, len, sr);
    buf.copyToChannel(mono, 0);
    const bs = oac.createBufferSource();
    bs.buffer = buf; bs.connect(oac.destination); bs.start(0);
    const rendered = await oac.startRendering();
    const out = rendered.getChannelData(0);
    let acc = 0; for (let i = 0; i < out.length; i++) acc += out[i] * out[i];
    const rms = Math.sqrt(acc / (out.length || 1));
    const wav = floatTo16BitWav(out, target);
    return { wav, rms };
  }, []);

  const analyzeVoice = useCallback(
    async (audioBlob: Blob): Promise<VoiceAnalysis> => {
      setIsAnalyzing(true);
      try {
        // Transcode to 16k WAV and send to backend
        const { wav, rms } = await webmToWav16k(audioBlob);
        const fd = new FormData();
        fd.append(
          "file",
          new File([wav], "recording.wav", { type: "audio/wav" })
        );

        const res = await fetch("/api/analyze", { method: "POST", body: fd });
        if (!res.ok) throw new Error(`Server error: ${res.status}`);
        const data: ApiResponse = await res.json();

        const lbl = (data.top?.label || "neutral").toLowerCase();
        const score = data.top?.score ?? 0;
        let sentiment: "positive" | "neutral" | "negative" = "neutral";
        if (["happy", "surprised"].includes(lbl)) sentiment = "positive";
        else if (["angry", "fearful", "sad", "disgust"].includes(lbl))
          sentiment = "negative";
        if (score < 0.25) sentiment = "neutral"; // lowâ€‘confidence safety

        const analysis: VoiceAnalysis = {
          emotion: data.top?.label || "neutral",
          confidence: score,
          sentiment,
          energy: energyPercent(rms),
          timestamp: new Date(),
        };
        return analysis;
      } finally {
        setIsAnalyzing(false);
      }
    },
    []
  );

  const monitorAudioLevel = useCallback(() => {
    if (!analyserRef.current) return;
    const analyser = analyserRef.current;
    const dataArray = new Uint8Array(analyser.frequencyBinCount);

    const updateLevel = () => {
      analyser.getByteFrequencyData(dataArray);
      const avg =
        dataArray.reduce((s, v) => s + v, 0) / (dataArray.length || 1);
      setAudioLevel(avg);
      if (isRecording) requestAnimationFrame(updateLevel);
    };
    updateLevel();
  }, [isRecording]);

  const startRecording = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          sampleRate: 44100,
        },
      });
      streamRef.current = stream;

      // audio analysis
      audioContextRef.current = new (window.AudioContext ||
        (window as any).webkitAudioContext)();
      analyserRef.current = audioContextRef.current.createAnalyser();
      const source = audioContextRef.current.createMediaStreamSource(stream);
      source.connect(analyserRef.current);
      analyserRef.current.fftSize = 256;

      // recorder
      const mediaRecorder = new MediaRecorder(stream, {
        mimeType: "audio/webm;codecs=opus",
      });
      const chunks: BlobPart[] = [];

      mediaRecorder.ondataavailable = (e) =>
        e.data.size > 0 && chunks.push(e.data);

      mediaRecorder.onstop = async () => {
        const audioBlob = new Blob(chunks, { type: "audio/webm" });
        const url = URL.createObjectURL(audioBlob);

        try {
          const analysis = await analyzeVoice(audioBlob);
          const rec: Recording = {
            id: Date.now().toString(),
            blob: audioBlob,
            url,
            duration: recordingTime,
            analysis,
            timestamp: new Date(),
          };
          setRecordings((prev) => [rec, ...prev]);
          setCurrentAnalysis(analysis);
          toast.success("Voice analysis complete!");
        } catch (e: any) {
          console.error(e);
          toast.error("Analysis failed. Is the backend running on :8000?");
          URL.revokeObjectURL(url);
        }
      };

      mediaRecorderRef.current = mediaRecorder;
      mediaRecorder.start();
      setIsRecording(true);
      setRecordingTime(0);

      recordingTimerRef.current = window.setInterval(
        () => setRecordingTime((p) => p + 1),
        1000
      );
      monitorAudioLevel();
      toast.success("Recording started!");
    } catch (err) {
      console.error("Mic error:", err);
      toast.error(
        "Microphone access denied. Please allow microphone permissions."
      );
    }
  };

  const stopRecording = () => {
    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();
      setIsRecording(false);
      setAudioLevel(0);
      if (recordingTimerRef.current)
        window.clearInterval(recordingTimerRef.current);
      if (streamRef.current)
        streamRef.current.getTracks().forEach((t) => t.stop());
      if (audioContextRef.current) audioContextRef.current.close();
    }
  };

  const playRecording = (rec: Recording) => {
    if (audioPlayerRef.current) audioPlayerRef.current.pause();
    const audio = new Audio(rec.url);
    audioPlayerRef.current = audio;
    audio.onplay = () => setIsPlaying(true);
    audio.onpause = () => setIsPlaying(false);
    audio.onended = () => setIsPlaying(false);
    audio.play().catch((e) => {
      console.error(e);
      toast.error("Failed to play recording");
    });
  };

  const pausePlayback = () => audioPlayerRef.current?.pause();

  const deleteRecording = (id: string) => {
    setRecordings((prev) => {
      const del = prev.find((r) => r.id === id);
      if (del) URL.revokeObjectURL(del.url);
      return prev.filter((r) => r.id !== id);
    });
    toast.success("Recording deleted");
  };

  const downloadRecording = (rec: Recording) => {
    const a = document.createElement("a");
    a.href = rec.url;
    a.download = `voice-recording-${rec.timestamp
      .toISOString()
      .slice(0, 19)}.webm`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
  };

  const formatTime = (s: number) =>
    `${Math.floor(s / 60)}:${(s % 60).toString().padStart(2, "0")}`;

  const getEmotionColor = (emotion: string) => {
    const colors: Record<string, string> = {
      happy: "bg-yellow-100 text-yellow-800 border-yellow-300",
      sad: "bg-blue-100 text-blue-800 border-blue-300",
      angry: "bg-red-100 text-red-800 border-red-300",
      surprised: "bg-purple-100 text-purple-800 border-purple-300",
      fearful: "bg-orange-100 text-orange-800 border-orange-300",
      neutral: "bg-gray-100 text-gray-800 border-gray-300",
      "excited / possibly angry": "bg-red-100 text-red-800 border-red-300",
      "sad / calm": "bg-blue-100 text-blue-800 border-blue-300",
    };
    return colors[emotion] || colors.neutral;
  };

  const getSentimentColor = (s: string) =>
    ((
      {
        positive: "text-green-600",
        negative: "text-red-600",
        neutral: "text-gray-600",
      } as any
    )[s] || "text-gray-600");

  useEffect(() => {
    return () => {
      stopRecording();
      recordings.forEach((r) => URL.revokeObjectURL(r.url));
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  /* ---------- UI ---------- */
  return (
    <div className="space-y-6">
      {/* Recorder */}
      <Card className="p-6 gradient-card shadow-emotion">
        <div className="text-center space-y-6">
          <div>
            <h3 className="text-xl font-semibold mb-2">
              Voice Emotion Analysis
            </h3>
            <p className="text-muted-foreground">
              Record your voice to analyze transcript, sentiment, and tone
            </p>
          </div>

          {/* Visualizer */}
          {isRecording && (
            <div className="space-y-2">
              <div className="flex justify-center">
                <div className="flex items-end gap-1 h-16">
                  {Array.from({ length: 10 }).map((_, i) => (
                    <div
                      key={i}
                      className="w-3 bg-primary rounded-t transition-all duration-150"
                      style={{
                        height: `${Math.max(
                          4,
                          (audioLevel / 255) *
                            64 *
                            (1 + Math.sin(Date.now() / 100 + i) * 0.5)
                        )}px`,
                      }}
                    />
                  ))}
                </div>
              </div>
              <p className="text-sm text-muted-foreground">
                Recording: {formatTime(recordingTime)}
              </p>
            </div>
          )}

          {/* Controls */}
          <div className="flex justify-center gap-4">
            <Button
              size="lg"
              variant={isRecording ? "destructive" : "default"}
              onClick={isRecording ? stopRecording : startRecording}
              className="transition-bounce shadow-glow"
              disabled={isAnalyzing}
            >
              {isRecording ? (
                <>
                  <Square className="h-5 w-5 mr-2" />
                  Stop Recording
                </>
              ) : (
                <>
                  <Mic className="h-5 w-5 mr-2" />
                  Start Recording
                </>
              )}
            </Button>

            {isPlaying && (
              <Button variant="outline" onClick={pausePlayback}>
                <Pause className="h-5 w-5 mr-2" />
                Pause
              </Button>
            )}
          </div>

          {isAnalyzing && (
            <p className="text-sm text-muted-foreground">Analyzingâ€¦</p>
          )}
        </div>
      </Card>

      {/* Current Analysis */}
      {currentAnalysis && (
        <Card className="p-6 gradient-card shadow-soft space-y-4">
          <h4 className="font-semibold">Latest Analysis Results</h4>

          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
            <div className="text-center">
              <p className="text-sm text-muted-foreground mb-1">Tone</p>
              <Badge
                className={`${getEmotionColor(currentAnalysis.emotion)} border`}
              >
                {currentAnalysis.emotion}
              </Badge>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground mb-1">Sentiment</p>
              <p
                className={`font-medium capitalize ${getSentimentColor(
                  currentAnalysis.sentiment
                )}`}
              >
                {currentAnalysis.sentiment}
              </p>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground mb-1">Confidence</p>
              <p className="font-medium">
                {Math.round(currentAnalysis.confidence * 100)}%
              </p>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground mb-1">Energy Level</p>
              <div className="space-y-1">
                <Progress value={currentAnalysis.energy} className="h-2" />
                <p className="text-sm font-medium">
                  {Math.round(currentAnalysis.energy)}%
                </p>
              </div>
            </div>
          </div>

          <div className="grid md:grid-cols-2 gap-4 text-sm">
            <div className="space-y-1">
              <p className="text-muted-foreground">Language</p>
              <p className="font-medium uppercase">
                {currentAnalysis.language}
              </p>
            </div>
            <div className="space-y-1">
              <p className="text-muted-foreground">Transcript</p>
              <p className="font-medium">{currentAnalysis.transcript}</p>
            </div>
            {currentAnalysis.prosody && (
              <>
                <div className="space-y-1">
                  <p className="text-muted-foreground">Pitch Î¼ (Hz)</p>
                  <p className="font-medium">
                    {currentAnalysis.prosody.pitch_mean_hz.toFixed(1)}
                  </p>
                </div>
                <div className="space-y-1">
                  <p className="text-muted-foreground">
                    Speech rate (frames/s)
                  </p>
                  <p className="font-medium">
                    {currentAnalysis.prosody.speech_rate_frames_per_sec.toFixed(
                      2
                    )}
                  </p>
                </div>
              </>
            )}
            {currentAnalysis.sentimentScores && (
              <div className="col-span-full">
                <p className="text-muted-foreground mb-1">Sentiment scores</p>
                <div className="flex gap-4">
                  <span>
                    ðŸ‘{" "}
                    {Math.round(currentAnalysis.sentimentScores.positive * 100)}
                    %
                  </span>
                  <span>
                    ðŸ˜{" "}
                    {Math.round(currentAnalysis.sentimentScores.neutral * 100)}%
                  </span>
                  <span>
                    ðŸ‘Ž{" "}
                    {Math.round(currentAnalysis.sentimentScores.negative * 100)}
                    %
                  </span>
                </div>
              </div>
            )}
          </div>
        </Card>
      )}

      {/* History */}
      {recordings.length > 0 && (
        <Card className="p-6 gradient-card shadow-soft">
          <h4 className="font-semibold mb-4">
            Voice Recordings ({recordings.length})
          </h4>
          <div className="space-y-3">
            {recordings.map((r) => (
              <div
                key={r.id}
                className="p-4 border rounded-lg bg-background/50"
              >
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-4">
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => playRecording(r)}
                    >
                      <Play className="h-4 w-4" />
                    </Button>

                    <div>
                      <p className="font-medium">
                        {r.timestamp.toLocaleDateString()} at{" "}
                        {r.timestamp.toLocaleTimeString()}
                      </p>
                      <p className="text-sm text-muted-foreground">
                        Duration: {formatTime(r.duration)}
                      </p>
                    </div>

                    <div className="flex gap-2">
                      <Badge
                        className={`${getEmotionColor(
                          r.analysis.emotion
                        )} border text-xs`}
                      >
                        {r.analysis.emotion}
                      </Badge>
                      <Badge variant="outline" className="text-xs">
                        {Math.round(r.analysis.confidence * 100)}%
                      </Badge>
                    </div>
                  </div>

                  <div className="flex gap-2">
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => downloadRecording(r)}
                    >
                      <Download className="h-4 w-4" />
                    </Button>
                    <Button
                      variant="destructive"
                      size="sm"
                      onClick={() => deleteRecording(r.id)}
                    >
                      <Trash2 className="h-4 w-4" />
                    </Button>
                  </div>
                </div>
              </div>
            ))}
          </div>
        </Card>
      )}
    </div>
  );
};
